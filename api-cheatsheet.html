fetch helpers

const API_URL = 'https://example.com/api';

async function getItems(signal?: AbortSignal): Promise<Item[]> {
  const r = await fetch(`${API_URL}/items`, { signal });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return r.json();
}

async function postItem(input: { title: string }): Promise<Item> {
  const r = await fetch(`${API_URL}/items`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return r.json();
}

Boundary validation (Zod)

Validate the response shape at the network boundary, not deep in the UI. parse throws on mismatch (use for internal API contracts); safeParse returns a result object (use for input you want to branch on).

import { z } from 'zod';

const ItemSchema = z.object({
  id: z.string(),
  title: z.string(),
  createdAt: z.number(),
});
const ItemsSchema = z.array(ItemSchema);

type Item = z.infer<typeof ItemSchema>;   // schema is the single source of truth for the type

async function getItems(signal?: AbortSignal): Promise<Item[]> {
  const r = await fetch(`${API_URL}/items`, { signal });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return ItemsSchema.parse(await r.json());   // throws -> RQ surfaces it as a query error
}

// safeParse when you want to branch instead of throw
const parsed = ItemSchema.safeParse(json);
if (!parsed.success) {
  console.warn(parsed.error.flatten());
  return fallback;
}
return parsed.data;

A schema mismatch becomes a normal query/mutation error, so the existing error UI handles it — no separate code path.

Optimistic update

const qc = useQueryClient();

const mutation = useMutation({
  mutationFn: postItem,

  onMutate: async (input) => {
    await qc.cancelQueries({ queryKey: ['items'] });
    const previous = qc.getQueryData<Item[]>(['items']);

    const optimistic: Item = {
      id: `temp-${Date.now()}`,
      title: input.title,
      createdAt: Date.now(),
    };
    qc.setQueryData<Item[]>(['items'], (old = []) => [optimistic, ...old]);

    return { previous };
  },

  onError: (_err, _input, ctx) => {
    if (ctx?.previous) qc.setQueryData(['items'], ctx.previous);
  },

  onSettled: () => {
    qc.invalidateQueries({ queryKey: ['items'] });
  },
});

Idempotency keys (mutation header)

Server-side dedupes retries / double-taps by client-generated UUID. Same key + same operation = same result, no double-processing.

function makeIdempotencyKey(): string {
  return 'idem-' + Date.now() + '-' + Math.random().toString(36).slice(2, 10);
}

async function sendCommand(input: { vehicleId: string; action: string }): Promise<Result> {
  const idempotencyKey = makeIdempotencyKey();
  const r = await fetch(`${API_URL}/command`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Idempotency-Key': idempotencyKey,
    },
    body: JSON.stringify(input),
  });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return r.json();
}

// Pair with useMutation — RQ's built-in retry will reuse the SAME key on retry
// because the mutation variables (input) are stable. Server treats retry as duplicate.
const mutation = useMutation({
  mutationFn: sendCommand,
  retry: 2,
});

For client-controlled keys that persist across re-renders, hoist the key generation into onMutate and store it in the context.

Token refresh on 401

Single in-flight refresh — all 401-failed requests await the same refresh promise.

let refreshPromise: Promise<string> | null = null;

async function getAccessToken(): Promise<string> {
  if (refreshPromise) return refreshPromise; // dedupe concurrent refreshes

  refreshPromise = (async () => {
    const refreshToken = await SecureStore.getItemAsync('refresh');
    const r = await fetch(`${API_URL}/auth/refresh`, {
      method: 'POST',
      body: JSON.stringify({ refreshToken }),
    });
    if (!r.ok) throw new Error('refresh failed');
    const { access } = await r.json();
    await SecureStore.setItemAsync('access', access);
    return access;
  })();

  try { return await refreshPromise; }
  finally { refreshPromise = null; }
}

async function authedFetch(url: string, init: RequestInit = {}): Promise<Response> {
  let token = await SecureStore.getItemAsync('access');
  let r = await fetch(url, {
    ...init,
    headers: { ...init.headers, Authorization: `Bearer ${token}` },
  });

  if (r.status === 401) {
    token = await getAccessToken();   // refresh once, then retry
    r = await fetch(url, {
      ...init,
      headers: { ...init.headers, Authorization: `Bearer ${token}` },
    });
  }
  return r;
}

Request deduplication (RQ built-in)

React Query already dedupes — multiple components calling useQuery with the same key share one in-flight request. This is one of the strongest reasons to use RQ over hand-rolled useEffect + fetch.

// Screen A and Screen B both mount and both call:
useQuery({ queryKey: ['user', userId], queryFn: () => getUser(userId) });

// → ONE network request. Both screens get the same data from the cache.
// → On `staleTime` expiration, both subscribe to the next refetch.

For non-RQ fetches that need dedup (e.g., custom hook), use a module-scoped Map<key, Promise> — same pattern as the token refresh above.

Cache strategy — invalidate vs setQueryData vs refetch

ActionUse when
invalidateQueriesServer is source of truth. Mark stale + refetch on next observer.
setQueryDataYou know the exact new shape (e.g., after a confirmed mutation response). Skips the network round-trip.
refetchQueriesForce immediate refetch (foreground), not just stale-marking.
removeQueriesCache should be wiped (logout, account switch).
resetQueriesReset to initial state (loading) and refetch.
 
// Confirmed mutation — directly write the response (saves a refetch)
onSuccess: (serverItem) => {
  qc.setQueryData<Item[]>(['items'], (old = []) =>
    old.map(i => i.id === serverItem.id ? serverItem : i)
  );
}

// Mutation that affects related queries — invalidate broadly
onSuccess: () => {
  qc.invalidateQueries({ queryKey: ['items'] });        // items list
  qc.invalidateQueries({ queryKey: ['itemCount'] });    // derived count
}

// Logout — wipe everything user-specific
const onLogout = () => {
  qc.removeQueries({ predicate: (q) => q.queryKey[0] === 'user' });
  qc.removeQueries({ predicate: (q) => q.queryKey[0] === 'items' });
};

Offline outbox queue — basic

Queue writes while offline, persist them, drain in order when connectivity returns. Each job carries a stable idempotency key generated once at enqueue, so a retry after a flaky connection can't double-apply.

import NetInfo from '@react-native-community/netinfo';
import { MMKV } from 'react-native-mmkv';

const storage = new MMKV();
const KEY = 'outbox';

type Job = {
  idempotencyKey: string;   // identity + sent on every retry
  url: string;
  method: 'POST' | 'PUT' | 'PATCH' | 'DELETE';
  body: unknown;
  attempts: number;
};

function load(): Job[] {
  const raw = storage.getString(KEY);
  return raw ? (JSON.parse(raw) as Job[]) : [];
}
function save(jobs: Job[]) {
  storage.set(KEY, JSON.stringify(jobs));
}

let draining = false;

export function enqueue(job: Omit<Job, 'attempts'>) {
  const jobs = load();
  jobs.push({ ...job, attempts: 0 });
  save(jobs);
  void drain();                       // try now; no-op if offline
}

export async function drain(): Promise<void> {
  if (draining) return;               // single drainer
  const net = await NetInfo.fetch();
  if (!net.isConnected) return;
  draining = true;
  try {
    let jobs = load();
    while (jobs.length) {
      const job = jobs[0];            // FIFO — preserve order
      try {
        const r = await fetch(job.url, {
          method: job.method,
          headers: {
            'Content-Type': 'application/json',
            'Idempotency-Key': job.idempotencyKey,
          },
          body: JSON.stringify(job.body),
        });
        if (!r.ok) throw new Error(`HTTP ${r.status}`);
        jobs = jobs.slice(1);         // success -> drop head
        save(jobs);
      } catch {
        jobs[0] = { ...job, attempts: job.attempts + 1 };
        save(jobs);
        break;                        // stop; retry on next reconnect
      }
    }
  } finally {
    draining = false;
  }
}

// Drain whenever connectivity is restored
NetInfo.addEventListener((state) => {
  if (state.isConnected) void drain();
});

The caller — a feature/handler — owns the request and mints the idempotency key once; the queue just delivers it:

// Caller: enqueue a write, and (optionally) update the cache optimistically.
function useAddItem() {
  const qc = useQueryClient();

  return (input: { title: string }) => {
    const tempId = `temp-${Date.now()}`;

    // optimistic cache write — UI updates immediately (same shape as the optimistic section)
    qc.setQueryData<Item[]>(['items'], (old = []) => [
      { id: tempId, title: input.title, createdAt: Date.now() },
      ...old,
    ]);

    // durable enqueue — survives offline + restart; tempId doubles as the idempotency key
    enqueue({
      idempotencyKey: tempId,        // minted ONCE here; drain reuses it on every retry
      url: `${API_URL}/items`,
      method: 'POST',
      body: input,
    });
  };
}

// in a component
const addItem = useAddItem();
<Pressable onPress={() => addItem({ title: 'New' })}><Text>Add</Text></Pressable>

Pair with an optimistic cache write so the UI updates immediately; the server reconciles when the job drains. Same idempotency key on replay means a double-send is deduped server-side. With the Zustand version below, the call is identical — just useOutbox.getState().enqueue(...) (or const enqueue = useOutbox((s) => s.enqueue) inside a component).

Why a BFF (backend-for-frontend)

One client-shaped endpoint instead of the app orchestrating N service calls.

ReasonWhat it buys the client
AggregationBFF fans out to many services, returns one response — the app makes a single round-trip (latency + battery win on cellular).
ShapingPayload trimmed to exactly what the screen needs — no over-fetch, smaller responses.
Version decouplingContract is owned per client, so the app payload evolves without touching core services; old app versions keep working.
Topology hidingClient never learns which services exist or how they're split.

Server-driven UI (component registry)

Server sends a tree of typed nodes; the client maps each type to a component. Unknown types render null, so a new server widget never crashes an old client — that is what makes server-driven rollout safe.

type UINode = {
  type: string;
  props?: Record<string, unknown>;
  children?: UINode[];
};

const registry: Record<string, React.FC<any>> = {
  text:   ({ value }) => <Text>{value}</Text>,
  button: ({ label, onPress }) => <Pressable onPress={onPress}><Text>{label}</Text></Pressable>,
  stack:  ({ children }) => <View>{children}</View>,
};

function renderNode(node: UINode): React.ReactNode {
  const Comp = registry[node.type];
  if (!Comp) return null;                 // unknown type -> ignore, don't crash
  const kids = node.children?.map((c, i) => (
    <React.Fragment key={i}>{renderNode(c)}</React.Fragment>
  ));
  return <Comp {...node.props}>{kids}</Comp>;
}

// Validate the layout payload at the boundary, then render:
const LayoutSchema: z.ZodType<UINode> = z.lazy(() =>
  z.object({
    type: z.string(),
    props: z.record(z.unknown()).optional(),
    children: z.array(LayoutSchema).optional(),
  })
);

function Screen({ json }: { json: unknown }) {
  const tree = LayoutSchema.parse(json);
  return <>{renderNode(tree)}</>;
}

Evolve additively: add new node types, never repurpose an existing one. Old clients ignore types they don't know; new clients render them.

useInfiniteQuery (pagination)

const {
  data, fetchNextPage, hasNextPage, isFetchingNextPage,
} = useInfiniteQuery<Page, Error>({
  queryKey: ['items', 'infinite'],
  queryFn: ({ pageParam, signal }) =>
    fetch(`${API_URL}/items?cursor=${pageParam ?? ''}`, { signal }).then(r => r.json()),
  initialPageParam: null as string | null,
  getNextPageParam: (last) => last.nextCursor,
});

const items = data?.pages.flatMap((p) => p.items) ?? [];

<FlatList
  data={items}
  keyExtractor={(i) => i.id}
  renderItem={({ item }) => <Row item={item} />}
  onEndReached={() => { if (hasNextPage && !isFetchingNextPage) fetchNextPage(); }}
  onEndReachedThreshold={0.5}
  ListFooterComponent={isFetchingNextPage ? <ActivityIndicator /> : null}
/>
