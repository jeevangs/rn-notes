Code-only reference for common RN + @tanstack/react-query patterns.

fetch helpers

const API_URL = 'https://example.com/api';

async function getItems(signal?: AbortSignal): Promise<Item[]> {
  const r = await fetch(`${API_URL}/items`, { signal });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return r.json();
}

async function postItem(input: { title: string }): Promise<Item> {
  const r = await fetch(`${API_URL}/items`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return r.json();
}

useInfiniteQuery (pagination)

const {
  data, fetchNextPage, hasNextPage, isFetchingNextPage,
} = useInfiniteQuery<Page, Error>({
  queryKey: ['items', 'infinite'],
  queryFn: ({ pageParam, signal }) =>
    fetch(`${API_URL}/items?cursor=${pageParam ?? ''}`, { signal }).then(r => r.json()),
  initialPageParam: null as string | null,
  getNextPageParam: (last) => last.nextCursor,
});

const items = data?.pages.flatMap((p) => p.items) ?? [];

<FlatList
  data={items}
  keyExtractor={(i) => i.id}
  renderItem={({ item }) => <Row item={item} />}
  onEndReached={() => { if (hasNextPage && !isFetchingNextPage) fetchNextPage(); }}
  onEndReachedThreshold={0.5}
  ListFooterComponent={isFetchingNextPage ? <ActivityIndicator /> : null}
/>

Optimistic update

const qc = useQueryClient();

const mutation = useMutation({
  mutationFn: postItem,

  onMutate: async (input) => {
    await qc.cancelQueries({ queryKey: ['items'] });
    const previous = qc.getQueryData<Item[]>(['items']);

    const optimistic: Item = {
      id: `temp-${Date.now()}`,
      title: input.title,
      createdAt: Date.now(),
    };
    qc.setQueryData<Item[]>(['items'], (old = []) => [optimistic, ...old]);

    return { previous };
  },

  onError: (_err, _input, ctx) => {
    if (ctx?.previous) qc.setQueryData(['items'], ctx.previous);
  },

  onSettled: () => {
    qc.invalidateQueries({ queryKey: ['items'] });
  },
});

Debounce hook

function useDebounced<T>(value: T, delayMs: number): T {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const id = setTimeout(() => setDebounced(value), delayMs);
    return () => clearTimeout(id);
  }, [value, delayMs]);
  return debounced;
}

Race-safe search

function SearchScreen() {
  const [search, setSearch] = useState('');
  const debouncedSearch = useDebounced(search, 300);

  const { data: items = [], isFetching } = useQuery<Item[], Error>({
    queryKey: ['items', 'search', debouncedSearch],
    queryFn: ({ signal }) =>
      fetch(`${API_URL}/search?q=${encodeURIComponent(debouncedSearch)}`, { signal })
        .then((r) => r.json()),
    enabled: debouncedSearch.length >= 2,
    placeholderData: (prev) => prev,
  });

  return (
    <View>
      <TextInput value={search} onChangeText={setSearch} placeholder="Search..." />
      {isFetching && <ActivityIndicator size="small" />}
      <FlatList data={items} keyExtractor={(i) => i.id} renderItem={({ item }) => <Text>{item.title}</Text>} />
    </View>
  );
}

WebSocket hook

Minimal: open / message / close + reconnect with backoff. Layer AppState / heartbeat as needed.

type WsStatus = 'idle' | 'connecting' | 'open' | 'closed';

function useWebSocket<T>(url: string, onMessage: (msg: T) => void) {
  const [status, setStatus] = useState<WsStatus>('idle');
  const wsRef = useRef<WebSocket | null>(null);
  const attemptsRef = useRef(0);
  const isMountedRef = useRef(true);
  const onMessageRef = useRef(onMessage);
  onMessageRef.current = onMessage;

  useEffect(() => {
    isMountedRef.current = true;

    const connect = () => {
      if (!isMountedRef.current) return;
      setStatus('connecting');
      const ws = new WebSocket(url);
      wsRef.current = ws;

      ws.onopen = () => { attemptsRef.current = 0; setStatus('open'); };
      ws.onmessage = (e) => {
        try { onMessageRef.current(JSON.parse(e.data) as T); } catch {}
      };
      ws.onerror = () => ws.close();
      ws.onclose = () => {
        setStatus('closed');
        if (!isMountedRef.current) return;
        const attempt = ++attemptsRef.current;
        const delay = Math.min(30_000, 1000 * 2 ** attempt) + Math.random() * 1000;
        setTimeout(connect, delay);
      };
    };

    connect();
    return () => {
      isMountedRef.current = false;
      wsRef.current?.close();
    };
  }, [url]);

  return { status };
}

FlatList — perf props

PropNotes
keyExtractorReturn a stable id, not the index. Required for memo of rows to stick.
getItemLayoutIf row height is fixed: (_, i) => ({ length: H, offset: H * i, index: i }). Skips measurement.
initialNumToRenderDefault 10. Drop on cold start if rows are heavy.
windowSizeDefault 21 (viewports). Reduce for memory; increase for fewer blanks while scrolling.
maxToRenderPerBatchHow many rows to render per batch. Lower = smoother scroll, slower fill.
removeClippedSubviewsDetaches off-screen native views. Android-friendly. Test for unmount bugs.
ListEmptyComponentRenders when data is empty — cheaper than conditional JSX above the list.

memo / useMemo / useCallback

memo on a child

const Row = memo(function Row({ item }: { item: Item }) {
  return (
    <View style={styles.row}>
      <Text>{item.title}</Text>
    </View>
  );
});

useCallback — only useful if the function is passed to a memoized child

const onPressRow = useCallback((id: string) => {
  navigation.navigate('Detail', { id });
}, [navigation]);

useMemo — for expensive derivations only

const filtered = useMemo(
  () => items.filter((i) => i.title.includes(query)),
  [items, query]
);

Both useMemo and useCallback re-create their cache each render based on deps. They cost CPU. Skip them when the work is cheap and the consumer is not memoized.

Full screen example

type Item = { id: string; title: string; createdAt: number };

const API_URL = 'https://example.com/api';

async function getItems(signal?: AbortSignal): Promise<Item[]> {
  const r = await fetch(`${API_URL}/items`, { signal });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return r.json();
}

async function postItem(input: { title: string }): Promise<Item> {
  const r = await fetch(`${API_URL}/items`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return r.json();
}

const Row = memo(function Row({ item, onPress }: {
  item: Item; onPress: (id: string) => void;
}) {
  return (
    <Pressable onPress={() => onPress(item.id)} style={styles.row}>
      <Text>{item.title}</Text>
    </Pressable>
  );
});

function Screen() {
  const qc = useQueryClient();
  const [draft, setDraft] = useState('');

  const { data = [], isLoading, isError, error, isRefetching, refetch } =
    useQuery<Item[], Error>({
      queryKey: ['items'],
      queryFn: ({ signal }) => getItems(signal),
    });

  const mutation = useMutation({
    mutationFn: postItem,
    onMutate: async (input) => {
      await qc.cancelQueries({ queryKey: ['items'] });
      const prev = qc.getQueryData<Item[]>(['items']);
      const optimistic: Item = {
        id: `temp-${Date.now()}`, title: input.title, createdAt: Date.now(),
      };
      qc.setQueryData<Item[]>(['items'], (old = []) => [optimistic, ...old]);
      return { prev };
    },
    onError: (_e, _i, ctx) => {
      if (ctx?.prev) qc.setQueryData(['items'], ctx.prev);
    },
    onSettled: () => qc.invalidateQueries({ queryKey: ['items'] }),
  });

  const onPressRow = useCallback((id: string) => {
    console.log('row', id);
  }, []);

  const renderItem = useCallback(
    ({ item }: ListRenderItemInfo<Item>) => <Row item={item} onPress={onPressRow} />,
    [onPressRow]
  );

  const keyExtractor = useCallback((i: Item) => i.id, []);

  if (isLoading) return <ActivityIndicator style={styles.center} />;
  if (isError)   return <Text style={styles.center}>{error.message}</Text>;

  return (
    <View style={styles.container}>
      <View style={styles.composer}>
        <TextInput
          value={draft}
          onChangeText={setDraft}
          placeholder="Add item..."
          style={styles.input}
        />
        <Pressable
          onPress={() => { mutation.mutate({ title: draft }); setDraft(''); }}
          disabled={!draft || mutation.isPending}
          style={styles.button}
        >
          <Text>Add</Text>
        </Pressable>
      </View>
      <FlatList
        data={data}
        renderItem={renderItem}
        keyExtractor={keyExtractor}
        refreshControl={
          <RefreshControl refreshing={isRefetching} onRefresh={() => refetch()} />
        }
        ListEmptyComponent={<Text>Nothing yet</Text>}
        initialNumToRender={10}
        windowSize={11}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  center: { flex: 1, textAlign: 'center', marginTop: 40 },
  composer: { flexDirection: 'row', padding: 12, gap: 8 },
  input: { flex: 1, borderWidth: 1, borderColor: '#ccc', padding: 8, borderRadius: 6 },
  button: { padding: 8, backgroundColor: '#eee', borderRadius: 6 },
  row: { padding: 12, borderBottomWidth: 1, borderBottomColor: '#eee' },
});
