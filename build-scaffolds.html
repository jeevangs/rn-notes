// (demo) mock socket — emits every 4s so live updates show:
function useChatSocket(roomId: string, key: readonly unknown[]) {
  const qc = useQueryClient();
  useEffect(() => {
    let n = 0;
    const id = setInterval(() => {
      qc.setQueryData<Message[]>(key, (old = []) =>
        mergeMessage(old, { id: `live-${n}`, roomId, senderId: 'them', text: `live ${n++}`, ts: Date.now() }));
    }, 4000);
    return () => clearInterval(id);
  }, [roomId]);
}

// ── useChat: the integration ──
function useChat(roomId: string) {
  const qc = useQueryClient();
  const key = ['messages', roomId] as const;

  const { data: messages = [], status } = useQuery({
    queryKey: key,
    queryFn: () => getMessages(roomId),
    select: (m) => [...m].sort((a, b) => b.ts - a.ts),
  });

  useChatSocket(roomId, key);

  const send = (text: string) => {
    const clientId = `c-${Date.now()}`;
    qc.setQueryData<Message[]>(key, (old = []) => [
      ...old,
      { id: `temp-${clientId}`, clientId, roomId, senderId: currentUserId, text, ts: Date.now(), status: 'pending' },
    ]);
    enqueue({ idempotencyKey: clientId, url: `${API_URL}/rooms/${roomId}/messages`, method: 'POST', body: { clientId, text } });
  };

  return { messages, send, status };
}

import React, { useState, useEffect, useRef, memo } from 'react';
import { View, Text, FlatList, StyleSheet } from 'react-native';
import { create } from 'zustand';
import { QueryClient, QueryClientProvider, useQuery } from '@tanstack/react-query';

// ── types ──
type Telemetry = { vehicleId: string; battery: number; speed: number; ts: number };

// ── config / mock backend (swap for real) ──
const API_URL = 'https://example.com/api';
const FLUSH_MS = 200;                          // 5Hz
const MOCK_IDS = ['veh-1', 'veh-2', 'veh-3'];

async function getBaseline(): Promise<Telemetry[]> {
  await new Promise((r) => setTimeout(r, 300));
  return MOCK_IDS.map((vehicleId) => ({ vehicleId, battery: 50 + Math.random() * 50, speed: 0, ts: Date.now() }));
}

// ── store: the source of truth (ts-guarded) ──
type TelemetryState = {
  byId: Record<string, Telemetry>;
  applyMany: (patches: Telemetry[]) => void;
};
const useStore = create<TelemetryState>((set) => ({
  byId: {},
  applyMany: (patches) => set((s) => {
    const next = { ...s.byId };
    for (const p of patches) {
      const prev = next[p.vehicleId];
      if (prev && prev.ts >= p.ts) continue;     // drop stale / out-of-order
      next[p.vehicleId] = p;
    }
    return { byId: next };                       // one set() per flush → one render pass
  }),
}));

// ── mechanisms ──
// (prod) your reconnect + backoff hook — signature only here:
declare function useWebSocket<T>(url: string, onMessage: (m: T) => void): { status: string };

// (prod) real wrapper validates frames + uses useWebSocket:
//   function useTelemetrySocket(onFrame) {
//     useWebSocket<unknown>('wss://example.com/telemetry', (raw) => {
//       const p = TelemetrySchema.safeParse(raw); if (p.success) onFrame(p.data);
//     });
//   }
// (demo) mock socket — emits ~30Hz so the 5Hz flush visibly coalesces:
function useTelemetrySocket(onFrame: (t: Telemetry) => void) {
  const onFrameRef = useRef(onFrame);
  onFrameRef.current = onFrame;
  useEffect(() => {
    const id = setInterval(() => {
      const vehicleId = MOCK_IDS[Math.floor(Math.random() * MOCK_IDS.length)];
      onFrameRef.current({ vehicleId, battery: 20 + Math.random() * 80, speed: Math.random() * 80, ts: Date.now() });
    }, 33);
    return () => clearInterval(id);
  }, []);
}

// ── ingest: coalesce in a ref buffer, flush to the store at 5Hz ──
function useTelemetryIngest() {
  const bufferRef = useRef<Record<string, Telemetry>>({});
  useTelemetrySocket((frame) => {
    const prev = bufferRef.current[frame.vehicleId];
    if (prev && prev.ts >= frame.ts) return;
    bufferRef.current[frame.vehicleId] = frame;     // non-reactive write → no render
  });
  useEffect(() => {
    const id = setInterval(() => {
      const batch = Object.values(bufferRef.current);
      if (batch.length === 0) return;
      bufferRef.current = {};
      useStore.getState().applyMany(batch);
    }, FLUSH_MS);
    return () => clearInterval(id);
  }, []);
}

// ── dashboard hook: seed + live ingest + id list ──
function useDashboard() {
  useQuery({ queryKey: ['baseline'], queryFn: getBaseline, staleTime: Infinity,
    onSuccess: (rows) => useStore.getState().applyMany(rows) });
  useTelemetryIngest();
  const ids = useStore((s) => Object.keys(s.byId));  // only changes when the set of ids changes
  return { ids };
}

// ── UI: ready ──
const Tile = memo(function Tile({ id }: { id: string }) {

  return (
    <View style={styles.tile}>

      <Text style={styles.metric}>{Math.round(t.battery)}%</Text>
      <Text style={styles.sub}>{Math.round(t.speed)} mph</Text>
    </View>
  );
});

function Dashboard() {

  return (

      numColumns={2}
      keyExtractor={(id) => id}
      contentContainerStyle={styles.grid}

// ── types ──
type Segment = { uri: string; duration?: number };
type Story = { id: string; authorName: string; avatarUri: string; segments: Segment[] };

const DEFAULT_DURATION = 5000;

// ── mock backend (swap for real getStories) ──
const PH = (seed: string, w = 1080, h = 1920) => `https://picsum.photos/seed/${seed}/${w}/${h}`;
const MOCK_STORIES: Story[] = [

const StoryTile = memo(function StoryTile({ story, onPress }: { story: Story; onPress: (id: string) => void }) {
  return (
    <Pressable style={styles.tile} onPress={() => onPress(story.id)}>
      <View style={styles.ring}><Image source={{ uri: story.avatarUri }} style={styles.avatar} /></View>
      <Text numberOfLines={1} style={styles.tileName}>{story.authorName}</Text>

});

// ── E · progress bars (derived from segmentIndex) ──
function ProgressBars({ count, index, progress }: { count: number; index: number; progress: number }) {
  return (
    <View style={styles.bars}>
      {Array.from({ length: count }, (_, i) => {
        const fill = i < index ? 1 : i === index ? progress : 0;
        return (
          <View key={i} style={styles.barTrack}>
            <View style={[styles.barFill, { flex: fill }]} />
            <View style={{ flex: 1 - fill }} />
          </View>
        );
      })}
    </View>
  );
}

// ── useStories: open/close + indices + timer + progress ──
function useStories(stories: Story[]) {
  const [openId, setOpenId] = useState<string | null>(null);
  const [storyIndex, setStoryIndex] = useState(0);
  const [segmentIndex, setSegmentIndex] = useState(0);
  const [paused, setPaused] = useState(false);
  const [progress, setProgress] = useState(0);

  const story = openId !== null ? stories[storyIndex] : null;
  const segment = story?.segments[segmentIndex];

  const open = (id: string) => {
    const idx = stories.findIndex((s) => s.id === id);
    if (idx < 0) return;
    setStoryIndex(idx); setSegmentIndex(0); setProgress(0); setOpenId(id);
  };
  const close = () => { setOpenId(null); setPaused(false); };

  const goNext = () => {
    if (!story) return;
    if (segmentIndex + 1 < story.segments.length) setSegmentIndex((i) => i + 1);
    else if (storyIndex + 1 < stories.length) { setStoryIndex((i) => i + 1); setSegmentIndex(0); }
    else close();
    setProgress(0);
  };
  const goPrev = () => {
    if (segmentIndex > 0) setSegmentIndex((i) => i - 1);
    else if (storyIndex > 0) { const p = stories[storyIndex - 1]; setStoryIndex((i) => i - 1); setSegmentIndex(p.segments.length - 1); }
    setProgress(0);
  };

  const goNextRef = useRef(goNext);
  goNextRef.current = goNext;

  useEffect(() => {
    if (openId === null || paused || !segment) return;
    const duration = segment.duration ?? DEFAULT_DURATION;
    const start = Date.now();
    setProgress(0);
    const tick = setInterval(() => {
      const p = Math.min(1, (Date.now() - start) / duration);
      setProgress(p);
      if (p >= 1) goNextRef.current();
    }, 50);
    return () => clearInterval(tick);                  // ◀ cleanup kills the stale timer
  }, [openId, storyIndex, segmentIndex, paused, segment]);

  useEffect(() => {
    const sub = AppState.addEventListener('change', (s) => setPaused(s !== 'active'));
    return () => sub.remove();
  }, []);

  return {
    openId, story, storyIndex, segmentIndex, segment, progress,
    open, close, goNext, goPrev,
    pause: () => setPaused(true), resume: () => setPaused(false),
  };
}

// ── screen: tile row + modal viewer ──
function StoriesScreen({ stories }: { stories: Story[] }) {
  const v = useStories(stories);
  return (

        keyExtractor={(s) => s.id}

        <View style={styles.viewer}>

          {v.story && <ProgressBars count={v.story.segments.length} index={v.segmentIndex} progress={v.progress} />}
          {v.story && (
            <View style={styles.header}>
              <Image source={{ uri: v.story.avatarUri }} style={styles.headerAvatar} />
              <Text style={styles.headerName}>{v.story.authorName}</Text>
              <Pressable style={styles.closeBtn} onPress={v.close}><Text style={styles.closeTxt}>✕</Text></Pressable>
            </View>
          )}

}

import React, { useState } from 'react';
import {
  View, Text, TextInput, Pressable, SafeAreaView, Alert, StyleSheet,
} from 'react-native';

// ── types ──
type SiteForm = { siteName: string; voltage: string; photoUris: string[] };
type StepKey = 'details' | 'measurements' | 'photos' | 'review';
const STEPS: StepKey[] = ['details', 'measurements', 'photos', 'review'];
const EMPTY_FORM: SiteForm = { siteName: '', voltage: '', photoUris: [] };

const API_URL = 'https://example.com/api';
const makeId = () => `a-${Date.now()}-${Math.floor(Math.random() * 1e6)}`;

// ── mechanism: offline enqueue — stubbed as Alert so submit is visible ──
type Job = { idempotencyKey: string; url: string; method: 'POST'; body: unknown };
function enqueue(job: Job): void {
  // (prod) persist to MMKV + drain on reconnect via NetInfo
  Alert.alert('Submitted', `Queued assessment (key ${job.idempotencyKey})`);
}

// ── validation gate (deepen → per-step Zod schema) ──
function canAdvance(step: StepKey, form: SiteForm): boolean {
  switch (step) {
    case 'details':      return form.siteName.trim().length > 0;
    case 'measurements': return Number(form.voltage) > 0;
    case 'photos':       return form.photoUris.length > 0;
    case 'review':       return true;
  }
}

// ── step views: dumb, controlled ──
type StepProps = { form: SiteForm; onChange: (patch: Partial<SiteForm>) => void };

function StepView({ step, form, onChange }: { step: StepKey } & StepProps) {
  switch (step) {
    case 'details':
      return (
        <View>
          <Text style={styles.label}>Site name</Text>
          <TextInput style={styles.input} value={form.siteName}
            onChangeText={(t) => onChange({ siteName: t })} placeholder="North lot array" />
        </View>
      );
    case 'measurements':
      return (
        <View>
          <Text style={styles.label}>Measured voltage (V)</Text>
          <TextInput style={styles.input} value={form.voltage} keyboardType="numeric"
            onChangeText={(t) => onChange({ voltage: t })} placeholder="240" />
        </View>
      );
    case 'photos':
      return (
        <View>
          <Text style={styles.label}>Photos ({form.photoUris.length})</Text>
          <Pressable style={styles.secondaryBtn}
            onPress={() => onChange({ photoUris: [...form.photoUris, `photo://${form.photoUris.length + 1}`] })}>
            <Text>Add photo</Text>
          </Pressable>
        </View>
      );
    case 'review':
      return (
        <View>
          <Text style={styles.label}>Review</Text>
          <Text>Site: {form.siteName || '—'}</Text>
          <Text>Voltage: {form.voltage || '—'}</Text>
          <Text>Photos: {form.photoUris.length}</Text>
        </View>
      );
  }
}

// ── progress + nav ──
function WizardProgress({ index }: { index: number }) {
  return (
    <View style={styles.progress}>
      {STEPS.map((key, i) => (
        <View key={key} style={[styles.segment, i <= index ? styles.segOn : styles.segOff]} />
      ))}
    </View>
  );
}

// ── useWizard: the integration ──
function useWizard() {
  const [index, setIndex] = useState(0);
  const [form, setForm] = useState<SiteForm>(EMPTY_FORM);
  const step = STEPS[index];
  const isLast = index === STEPS.length - 1;

  const update = (patch: Partial<SiteForm>) => setForm((f) => ({ ...f, ...patch }));
  const next = () => { if (canAdvance(step, form)) setIndex((i) => Math.min(i + 1, STEPS.length - 1)); };
  const back = () => setIndex((i) => Math.max(i - 1, 0));
  const submit = () => {
    const payload = { ...form, voltage: Number(form.voltage) };
    enqueue({ idempotencyKey: makeId(), url: `${API_URL}/assessments`, method: 'POST', body: payload });
  };

  return { index, step, isLast, form, update, next, back, submit, canNext: canAdvance(step, form) };
}

import React, { useState, useEffect, memo } from 'react';
import {
  View, Text, TextInput, FlatList, ActivityIndicator,
  RefreshControl, SafeAreaView, StyleSheet,
} from 'react-native';
import {
  QueryClient, QueryClientProvider, useInfiniteQuery,
} from '@tanstack/react-query';

// ── types ──
type Item = { id: string; title: string };
type Page = { items: Item[]; nextCursor: string | null };

// ── mock backend (swap for real) ──
const PAGE_SIZE = 20;
const ALL: Item[] = Array.from({ length: 200 }, (_, i) => ({ id: String(i), title: `Item ${i}` }));

// (prod) add a `signal?: AbortSignal` param and thread it into fetch so key changes abort.
async function getItems(cursor: string | null, q: string): Promise<Page> {
  await new Promise((r) => setTimeout(r, 300));
  const filtered = q ? ALL.filter((it) => it.title.toLowerCase().includes(q.toLowerCase())) : ALL;
  const start = cursor ? Number(cursor) : 0;
  const slice = filtered.slice(start, start + PAGE_SIZE);
  const next = start + PAGE_SIZE;
  return { items: slice, nextCursor: next < filtered.length ? String(next) : null };
}

// ── mechanism: debounce ──
function useDebounced<T>(value: T, delay = 300): T {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const id = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(id);
  }, [value, delay]);
  return debounced;
}

// ── useFeed: the integration ──
function useFeed() {
  const [search, setSearch] = useState('');
  const debouncedSearch = useDebounced(search, 300);

  const {
    data, fetchNextPage, hasNextPage, isFetchingNextPage,
    isRefetching, refetch, status,
  } = useInfiniteQuery<Page, Error>({
    queryKey: ['items', debouncedSearch],
    queryFn: ({ pageParam }) => getItems(pageParam as string | null, debouncedSearch),
    initialPageParam: null,
    getNextPageParam: (last) => last.nextCursor,
    enabled: debouncedSearch.length === 0 || debouncedSearch.length >= 2,
    placeholderData: (prev) => prev,
  });

  const items = data?.pages.flatMap((p) => p.items) ?? [];
  const loadMore = () => { if (hasNextPage && !isFetchingNextPage) void fetchNextPage(); };

  return { items, search, setSearch, status, loadMore, isFetchingNextPage, refetch, isRefetching };
}

// ── UI: ready ──
const Row = memo(function Row({ item }: { item: Item }) {
  return (
    <View style={styles.row}>
      <Text style={styles.title}>{item.title}</Text>
    </View>
  );
});

function FeedScreen() {
  const {
    items, search, setSearch, status,
    loadMore, isFetchingNextPage, refetch, isRefetching,
  } = useFeed();

  return (
    <SafeAreaView style={{ flex: 1 }}>
      <TextInput
        value={search}
        onChangeText={setSearch}
        placeholder="Search…"
        style={styles.search}
        autoCorrect={false}
        autoCapitalize="none"
      />
      {status === 'pending' ? (
        <ActivityIndicator style={{ flex: 1 }} />
      ) : (
        <FlatList
          data={items}
          keyExtractor={(it) => it.id}
          renderItem={({ item }) => <Row item={item} />}
          onEndReachedThreshold={0.5}
          onEndReached={loadMore}
          refreshControl={
            <RefreshControl refreshing={isRefetching} onRefresh={() => void refetch()} />
          }
          ListFooterComponent={isFetchingNextPage ? <ActivityIndicator style={{ margin: 16 }} /> : null}
          ListEmptyComponent={<Text style={styles.empty}>No results</Text>}
        />
      )}
    </SafeAreaView>
  );
}

const queryClient = new QueryClient();
export default function App() {
  return (
    <QueryClientProvider client={queryClient}>
      <FeedScreen />
    </QueryClientProvider>
  );
}
