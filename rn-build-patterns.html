Composer (input + send button)

function Composer({ onSend }: { onSend: (text: string) => void }) {
  const [draft, setDraft] = useState('');
  const canSend = draft.trim().length > 0;

  const handleSend = () => {
    if (!canSend) return;
    onSend(draft.trim());
    setDraft('');
  };

  return (
    <View style={styles.composer}>
      <TextInput
        value={draft}
        onChangeText={setDraft}
        placeholder="Message..."
        style={styles.input}
        multiline
        maxLength={2000}
        returnKeyType="send"
        onSubmitEditing={handleSend}
        blurOnSubmit={false}
        accessibilityLabel="Message input"
      />
      <Pressable
        onPress={handleSend}
        disabled={!canSend}
        style={[styles.sendButton, !canSend && styles.sendButtonDisabled]}
        accessibilityRole="button"
        accessibilityLabel="Send message"
      >
        <Text style={styles.sendButtonText}>Send</Text>
      </Pressable>
    </View>
  );
}

  composer: {

Full chat screen (composition)

export default function ChatScreen() {
  const [messages, setMessages] = useState<Message[]>(INITIAL_MESSAGES);

  const onSend = useCallback((text: string) => {
    const optimistic: Message = {
      id: `temp-${Date.now()}`,
      text,
      sender: 'me',
      timestamp: Date.now(),
      status: 'sending',
    };
    setMessages((prev) => [...prev, optimistic]);

    sendMessage(text)
      .then(({ id }) => {
        setMessages((prev) =>
          prev.map((m) => m.id === optimistic.id ? { ...m, id, status: 'sent' } : m)
        );
      })
      .catch(() => {
        setMessages((prev) =>
          prev.map((m) => m.id === optimistic.id ? { ...m, status: 'failed' } : m)
        );
      });
  }, []);

  return (
    <SafeAreaView style={{ flex: 1 }} edges={['top']}>
      <KeyboardAvoidingView
        behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
        style={{ flex: 1 }}
        keyboardVerticalOffset={Platform.OS === 'ios' ? 64 : 0}
      >
        <FlatList
          data={[...messages].reverse()}
          inverted
          renderItem={({ item }) => <MessageBubble message={item} />}
          keyExtractor={(m) => m.id}
          ListEmptyComponent={<Text style={styles.empty}>No messages yet</Text>}
        />
        <Composer onSend={onSend} />
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

Fullscreen viewer with auto-advance

function StoryViewer({ story, onClose }: { story: Story; onClose: () => void }) {
  const [segmentIndex, setSegmentIndex] = useState(0);
  const segment = story.segments[segmentIndex];

  useEffect(() => {
    if (!segment) { onClose(); return; }
    const id = setTimeout(() => setSegmentIndex((i) => i + 1), segment.duration);
    return () => clearTimeout(id);
  }, [segment, onClose]);

  const goNext = () => setSegmentIndex((i) => i + 1);
  const goPrev = () => setSegmentIndex((i) => Math.max(0, i - 1));

  if (!segment) return null;

  return (
    <Modal visible animationType="fade" onRequestClose={onClose}>
      <View style={styles.viewerContainer}>
        <Image source={{ uri: segment.uri }} style={styles.viewerImage} resizeMode="cover" />

        <ProgressBar segments={story.segments.length} current={segmentIndex} />

        <Pressable style={styles.tapLeft}  onPress={goPrev} />
        <Pressable style={styles.tapRight} onPress={goNext} />

        <Pressable style={styles.closeBtn} onPress={onClose}>
          <Text style={styles.closeText}>✕</Text>
        </Pressable>
      </View>
    </Modal>
  );
}

Validation pattern

type FormErrors = Partial<Record<'email' | 'password', string>>;

function validate(values: { email: string; password: string }): FormErrors {
  const errors: FormErrors = {};
  if (!values.email.includes('@')) errors.email = 'Invalid email';
  if (values.password.length < 8)  errors.password = 'Password too short';
  return errors;
}

const [touched, setTouched] = useState<Record<string, boolean>>({});
const errors = validate({ email, password });

<TextInput
  value={email}
  onChangeText={setEmail}
  onBlur={() => setTouched((t) => ({ ...t, email: true }))}
/>
{touched.email && errors.email && <Text style={styles.error}>{errors.email}</Text>}

Pressable with double-tap debounce

function useDebouncedPress(handler: () => void, ms = 500) {
  const lastTapRef = useRef(0);
  return useCallback(() => {
    const now = Date.now();
    if (now - lastTapRef.current < ms) return;  // ignore rapid double-tap
    lastTapRef.current = now;
    handler();
  }, [handler, ms]);
}

const onConfirmedPress = useDebouncedPress(onPress, 800);
<Pressable onPress={onConfirmedPress} />

State decision — local vs Zustand vs React Query

State typeWhere it lives
Form input, modal open/closed, currently-selected tabLocal useState — ephemeral, single-component
Theme, current user, app-wide preferences, auth stateZustand (or Context) — shared across screens, not server-derived
Anything that came from a server: vehicles, messages, telemetryReact Query — caching, dedupe, refetch, optimistic all built in
Ephemeral animation values (scroll position, gesture progress)Reanimated shared values — runs on UI thread

Anti-pattern: storing server data in Zustand then manually keeping it in sync with refetches. That's reinventing React Query.

MMKV (synchronous key-value store)

react-native-mmkv — a fast, synchronous, encrypted-optional key-value store built on Tencent's MMKV (C++ library backed by mmap). The reflexive replacement for AsyncStorage whenever you can afford to install a native module.

When to use it

  • Auth tokens — synchronous reads mean no useEffect dance to load on app start. Pair with the encryptionKey option (use a key from Keychain/Keystore).
  • App preferences / settings — theme, language, feature flags, last-viewed screen.
  • Small caches — last-fetched user profile, recently-viewed items, draft form values.
  • Hydration of Zustand or other stores on startup — synchronous reads mean the store is populated before the first render.

When NOT to use it

  • Large blobs (images, video, multi-MB JSON) — use the filesystem (expo-file-system) instead. MMKV reads the whole value into memory.
  • Relational data with queries — use SQLite (expo-sqlite, op-sqlite) or a real DB.
  • Anything cross-process (widgets, share extensions) — needs a more elaborate IPC setup.

Basic API

import { MMKV } from 'react-native-mmkv';

export const storage = new MMKV({
  id: 'app-default',                              // instance namespace
  encryptionKey: 'derived-from-keychain-or-keystore',  // optional
});

// Synchronous reads/writes — no await
storage.set('user.id', '12345');
const userId = storage.getString('user.id');     // '12345'

storage.set('darkMode', true);
const dark = storage.getBoolean('darkMode');     // true

storage.set('count', 42);
const count = storage.getNumber('count');        // 42

storage.delete('user.id');
storage.clearAll();

// Object/array — JSON encode yourself
storage.set('user', JSON.stringify({ id: '12345', name: 'Jeev' }));
const user = JSON.parse(storage.getString('user') ?? 'null');

Why interviewers ask about it

If you say "AsyncStorage" in a Staff round, expect: "Why not MMKV?" The answer they want: "Synchronous, mmap-backed, ~30× faster than AsyncStorage's SQLite. For auth tokens and preferences I'd default to MMKV unless we needed cross-process access or large blobs."

JS thread vs UI thread — what runs where

Most "ground-breaking" RN libraries since 2024 share one ambition: move work off the JS thread onto the UI thread so animations and gestures stay 60fps even when JS is busy.

WorkThreadNotes
Your component renders (React.render, hooks, JSX)JS threadBlocking JS = no new renders. Profile with React DevTools.
Animated API with useNativeDriver: trueUI threadOnly works for transform + opacity
Reanimated worklets ('worklet' directive)UI threadJS compiled to run synchronously on UI thread — full expressiveness
Gesture Handler eventsUI threadNative gesture recognizers; pair with Reanimated worklets
Skia useDerivedValue rendersUI threadCanvas redraws on UI thread, decoupled from React render
@expo/ui componentsNative (SwiftUI / Compose)Truly native rendering — no Yoga, no JSX bridge per frame
FlatList scroll (without virtualization tuning)JS + UI mixedScroll is native; cell rendering is JS — see FlatList perf props
Network / JSON.parseJS threadLarge parses can drop frames; offload to InteractionManager if needed

The Staff-level framing: "If it animates at 60fps or responds to a gesture, put it on the UI thread. If it computes business logic or talks to the network, JS thread. Bridge crossings are the cost."

@expo/ui — native SwiftUI / Jetpack Compose primitives

@expo/ui wraps platform-native primitives (SwiftUI on iOS, Jetpack Compose on Android) so they render on the platform's native UI pipeline — not through Yoga + React Native's view hierarchy. Released by Expo in 2025 as part of their "native everywhere" push.

What's in the box

  • <Button>, <Switch>, <Picker>, <Slider>, <DateTimePicker>, <ColorPicker>, <Section>, <List>
  • Components match platform conventions automatically (iOS rounded segmented controls, Android Material 3 chips, etc.)
  • Smaller bundle impact than equivalent RN components — no JS implementation, just declarative bridges

Why use it

  • Platform consistency without writing two implementations — the picker LOOKS like a picker on each platform
  • Animations / interactions handled by the OS — no JS thread involvement for the standard interactions
  • Accessibility free — system controls inherit OS-level a11y for free

Basic usage

        value={darkMode}
        onValueChange={setDarkMode}
      />
      <Picker
        label="Language"
        options={[{ label: 'English', value: 'en' }, { label: 'Español', value: 'es' }]}
        selected={language}
        onChange={setLanguage}
      />
      <Button
        title="Save"
        variant="filled"
        onPress={save}
      />
    </Section>
  );
}

When NOT to use it

  • Heavily branded UIs where the system look conflicts with your design system — stick with custom RN components
  • If you need consistent pixel-perfect behavior across both platforms (it's intentionally NOT pixel-identical)
  • Older RN versions / non-Expo workflows where the codegen hooks aren't available

Reanimated worklets — animations on the UI thread

The 'worklet' directive marks a JS function that compiles to run on the UI thread instead of the JS thread. Animations driven by shared values stay 60fps even if JS is doing heavy work elsewhere.

Core hooks

import Animated, {
  useSharedValue, useAnimatedStyle, withTiming, withSpring, Easing,
  useDerivedValue, runOnJS,
} from 'react-native-reanimated';

function FadeIn({ children }: { children: React.ReactNode }) {
  const opacity = useSharedValue(0);

  useEffect(() => {
    opacity.value = withTiming(1, {
      duration: 400,
      easing: Easing.inOut(Easing.cubic),
    });
  }, []);

  const animatedStyle = useAnimatedStyle(() => ({
    opacity: opacity.value,
    transform: [{ translateY: (1 - opacity.value) * 20 }],
  }));

  return <Animated.View style={animatedStyle}>{children}</Animated.View>;
}

Gesture-driven worklet (drag)

import { Gesture, GestureDetector } from 'react-native-gesture-handler';

function DraggableCard() {
  const translateX = useSharedValue(0);
  const translateY = useSharedValue(0);

  const pan = Gesture.Pan()
    .onUpdate((e) => {
      'worklet';                       // runs on UI thread synchronously
      translateX.value = e.translationX;
      translateY.value = e.translationY;
    })
    .onEnd(() => {
      'worklet';
      translateX.value = withSpring(0);
      translateY.value = withSpring(0);
    });

  const style = useAnimatedStyle(() => ({
    transform: [{ translateX: translateX.value }, { translateY: translateY.value }],
  }));

  return (
    <GestureDetector gesture={pan}>
      <Animated.View style={[styles.card, style]} />
    </GestureDetector>
  );
}

Calling back to JS thread when needed

const handleComplete = () => {
  Alert.alert('Done');                 // JS thread API — needs runOnJS
};

const tap = Gesture.Tap().onEnd(() => {
  'worklet';
  opacity.value = withTiming(0, { duration: 200 }, (finished) => {
    'worklet';
    if (finished) runOnJS(handleComplete)();   // bridge back to JS
  });
});

Common pitfalls

  • Forgetting 'worklet' in a callback passed to a gesture handler → silently runs on JS thread → janky gesture
  • Closing over non-shared values inside a worklet → "value is undefined on UI thread" — only shared values (useSharedValue) are visible
  • Calling React state setters directly from a worklet → must go through runOnJS(setState)(value)
  • Animating width / height / top / left instead of transform — triggers Yoga layout on each frame; expensive even on UI thread

React Native Skia — UI-thread canvas

@shopify/react-native-skia binds React Native to Skia (Google's 2D graphics engine — powers Chrome and Flutter). Draws on a canvas on the UI thread; renders are independent of React's render cycle when paired with useDerivedValue.

When to reach for it

  • Custom charts (line graphs, gauges, radial visualizations) — performant at thousands of points
  • Complex animations with multiple synchronized elements (e.g., a battery-charge gauge with a sweeping needle + glow + particles)
  • Image filters and effects — blur, gradients, masks, blend modes
  • Visualizations that update at telemetry rate — pair with Reanimated shared values for 60fps updates without re-renders

Minimal example — animated circle

import { Canvas, Circle, Group } from '@shopify/react-native-skia';
import { useSharedValue, withRepeat, withTiming } from 'react-native-reanimated';

function Pulse() {
  const r = useSharedValue(20);

  useEffect(() => {
    r.value = withRepeat(withTiming(60, { duration: 800 }), -1, true);
  }, []);

  return (
    <Canvas style={{ width: 200, height: 200 }}>
      <Group>
        <Circle cx={100} cy={100} r={r} color="rgba(0, 132, 255, 0.6)" />
      </Group>
    </Canvas>
  );
}

Use cases for Tesla Energy: live solar-production gauge, Megapack site status overlay, battery-charge radial, telemetry sparkline that updates 5Hz without re-rendering React.

The new architecture (Fabric + TurboModules)

The 2024-2025 rewrite of RN's internals. Default on for new RN apps in 2025+. Old: JS↔native talked through an async serialized bridge. New: JS Interface (JSI) lets JS call native (and vice versa) directly via shared memory — no serialization, no async.

What changes

  • Fabric (renderer): synchronous, multi-threaded. Layout can happen on the UI thread; React commits are dispatched without queuing across the bridge.
  • TurboModules: lazy-loaded native modules. Methods are typed (via codegen from TS interfaces). Most native calls become synchronous and zero-copy.
  • JSI (JavaScript Interface): the substrate everything sits on. JS objects can hold direct references to C++ HostObjects.
  • Codegen: Generate native interface code from TypeScript spec files (NativeSomething.ts) — fewer hand-written bridges.

Interview talking points

  • "In the new architecture, calling a native module is a synchronous JSI call, not a posted message across the bridge. That's why MMKV is synchronous out of the box — there's no async serialization layer to await."
  • "Fabric lets the renderer commit React updates without round-tripping through a queue. Animations and layout can stay synchronized with native, which is why Reanimated v3 can do interpolation on the UI thread without state divergence."
  • "Old arch is in maintenance — Meta's recommendation is opt into the new architecture for any new project. If we're on RN 0.76+, it's default."
