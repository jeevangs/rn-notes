SafeAreaView

Use react-native-safe-area-context (more reliable than the deprecated core SafeAreaView).

        <Screen />
      </SafeAreaView>
    </SafeAreaProvider>
  );
}

Message bubble (memoized)

const MessageBubble = memo(function MessageBubble({ message }: { message: Message }) {
  const isMine = message.sender === 'me';
  return (
    <View style={[styles.bubbleRow, isMine ? styles.bubbleRowMine : styles.bubbleRowThem]}>
      <View style={[styles.bubble, isMine ? styles.bubbleMine : styles.bubbleThem]}>
        <Text style={[styles.bubbleText, isMine ? styles.bubbleTextMine : null]}>
          {message.text}
        </Text>
        {message.status === 'sending' && <Text style={styles.statusText}>Sending...</Text>}
        {message.status === 'failed'  && <Text style={styles.statusError}>Failed - tap to retry</Text>}
      </View>
    </View>
  );
});

const styles = StyleSheet.create({
  bubbleRow:      { paddingHorizontal: 12, marginVertical: 2 },
  bubbleRowMine:  { alignItems: 'flex-end' },
  bubbleRowThem:  { alignItems: 'flex-start' },
  bubble:         { maxWidth: '78%', padding: 10, borderRadius: 16 },
  bubbleMine:     { backgroundColor: '#0084ff', borderBottomRightRadius: 4 },
  bubbleThem:     { backgroundColor: '#e9e9eb', borderBottomLeftRadius: 4 },
  bubbleText:     { fontSize: 15, color: '#000' },
  bubbleTextMine: { color: '#fff' },
  statusText:     { fontSize: 11, color: '#fff8', marginTop: 2 },

Inverted FlatList for chat

inverted flips the list so newest is at the bottom and the scroll position naturally stays at the latest message. Pass messages in reverse chronological order.

const renderItem = useCallback(
  ({ item }: ListRenderItemInfo<Message>) => <MessageBubble message={item} />,
  []
);

const keyExtractor = useCallback((m: Message) => m.id, []);

// messages must be in reverse-chronological order: newest first
const orderedMessages = useMemo(
  () => [...messages].sort((a, b) => b.timestamp - a.timestamp),
  [messages]
);

  renderItem={renderItem}

  showsVerticalScrollIndicator={false}

Composer (input + send button)

function Composer({ onSend }: { onSend: (text: string) => void }) {
  const [draft, setDraft] = useState('');
  const canSend = draft.trim().length > 0;

  const handleSend = () => {
    if (!canSend) return;
    onSend(draft.trim());
    setDraft('');
  };

  return (
    <View style={styles.composer}>
      <TextInput
        value={draft}
        onChangeText={setDraft}
        placeholder="Message..."
        style={styles.input}
        multiline
        maxLength={2000}
        returnKeyType="send"
        onSubmitEditing={handleSend}
        blurOnSubmit={false}
        accessibilityLabel="Message input"
      />
      <Pressable
        onPress={handleSend}
        disabled={!canSend}
        style={[styles.sendButton, !canSend && styles.sendButtonDisabled]}
        accessibilityRole="button"
        accessibilityLabel="Send message"
      >
        <Text style={styles.sendButtonText}>Send</Text>
      </Pressable>
    </View>
  );
}

const styles = StyleSheet.create({
  composer: {
    flexDirection: 'row', alignItems: 'flex-end',
    padding: 8, gap: 8, borderTopWidth: 1, borderTopColor: '#eee',
  },
  input: {
    flex: 1, minHeight: 40, maxHeight: 100,
    borderWidth: 1, borderColor: '#ccc', borderRadius: 18,
    paddingHorizontal: 14, paddingVertical: 8, fontSize: 15,
  },
  sendButton: {
    backgroundColor: '#0084ff', paddingHorizontal: 18, paddingVertical: 10,
    borderRadius: 18,
  },
  sendButtonDisabled: { backgroundColor: '#ccc' },
  sendButtonText: { color: '#fff', fontWeight: '600' },
});

Full chat screen (composition)

export default function ChatScreen() {
  const [messages, setMessages] = useState<Message[]>(INITIAL_MESSAGES);

  const onSend = useCallback((text: string) => {
    const optimistic: Message = {
      id: `temp-${Date.now()}`,
      text,
      sender: 'me',
      timestamp: Date.now(),
      status: 'sending',
    };
    setMessages((prev) => [...prev, optimistic]);

    sendMessage(text)
      .then(({ id }) => {
        setMessages((prev) =>
          prev.map((m) => m.id === optimistic.id ? { ...m, id, status: 'sent' } : m)
        );
      })
      .catch(() => {
        setMessages((prev) =>
          prev.map((m) => m.id === optimistic.id ? { ...m, status: 'failed' } : m)
        );
      });
  }, []);

  return (
    <SafeAreaView style={{ flex: 1 }} edges={['top']}>
      <KeyboardAvoidingView
        behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
        style={{ flex: 1 }}
        keyboardVerticalOffset={Platform.OS === 'ios' ? 64 : 0}
      >
        <FlatList
          data={[...messages].reverse()}
          inverted
          renderItem={({ item }) => <MessageBubble message={item} />}
          keyExtractor={(m) => m.id}
          ListEmptyComponent={<Text style={styles.empty}>No messages yet</Text>}
        />
        <Composer onSend={onSend} />
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

Fullscreen viewer with auto-advance

function StoryViewer({ story, onClose }: { story: Story; onClose: () => void }) {
  const [segmentIndex, setSegmentIndex] = useState(0);
  const segment = story.segments[segmentIndex];

  useEffect(() => {
    if (!segment) { onClose(); return; }
    const id = setTimeout(() => setSegmentIndex((i) => i + 1), segment.duration);
    return () => clearTimeout(id);
  }, [segment, onClose]);

  const goNext = () => setSegmentIndex((i) => i + 1);
  const goPrev = () => setSegmentIndex((i) => Math.max(0, i - 1));

  if (!segment) return null;

  return (
    <Modal visible animationType="fade" onRequestClose={onClose}>
      <View style={styles.viewerContainer}>
        <Image source={{ uri: segment.uri }} style={styles.viewerImage} resizeMode="cover" />

        <ProgressBar segments={story.segments.length} current={segmentIndex} />

        <Pressable style={styles.tapLeft}  onPress={goPrev} />
        <Pressable style={styles.tapRight} onPress={goNext} />

        <Pressable style={styles.closeBtn} onPress={onClose}>
          <Text style={styles.closeText}>✕</Text>
        </Pressable>
      </View>
    </Modal>
  );
}

Progress bar (segmented)

function ProgressBar({ segments, current }: { segments: number; current: number }) {
  return (
    <View style={styles.progressContainer}>
      {Array.from({ length: segments }, (_, i) => (
        <View key={i} style={styles.progressTrack}>
          <View
            style={[
              styles.progressFill,
              { width: i < current ? '100%' : i === current ? '50%' : '0%' },
            ]}
          />
        </View>
      ))}
    </View>
  );
}

const styles = StyleSheet.create({
  progressContainer: {
    flexDirection: 'row', position: 'absolute', top: 12, left: 12, right: 12, gap: 4,
  },
  progressTrack: { flex: 1, height: 3, backgroundColor: '#fff4', borderRadius: 2 },
  progressFill:  { height: '100%', backgroundColor: '#fff', borderRadius: 2 },
});

Validation pattern

type FormErrors = Partial<Record<'email' | 'password', string>>;

function validate(values: { email: string; password: string }): FormErrors {
  const errors: FormErrors = {};
  if (!values.email.includes('@')) errors.email = 'Invalid email';
  if (values.password.length < 8)  errors.password = 'Password too short';
  return errors;
}

const [touched, setTouched] = useState<Record<string, boolean>>({});
const errors = validate({ email, password });

<TextInput
  value={email}
  onChangeText={setEmail}
  onBlur={() => setTouched((t) => ({ ...t, email: true }))}
/>
{touched.email && errors.email && <Text style={styles.error}>{errors.email}</Text>}

Submit with disabled state

const hasErrors = Object.keys(errors).length > 0;
const [isSubmitting, setIsSubmitting] = useState(false);

const onSubmit = async () => {
  if (hasErrors) {
    setTouched({ email: true, password: true });  // force-show errors
    return;
  }
  setIsSubmitting(true);
  try { await login({ email, password }); }
  catch (e) { Alert.alert('Login failed'); }
  finally { setIsSubmitting(false); }
};

<Pressable
  onPress={onSubmit}
  disabled={hasErrors || isSubmitting}
  style={[styles.button, (hasErrors || isSubmitting) && styles.buttonDisabled]}
>
  {isSubmitting
    ? <ActivityIndicator color="#fff" />
    : <Text style={styles.buttonText}>Log in</Text>}
</Pressable>

Pressable with double-tap debounce

function useDebouncedPress(handler: () => void, ms = 500) {
  const lastTapRef = useRef(0);
  return useCallback(() => {
    const now = Date.now();
    if (now - lastTapRef.current < ms) return;  // ignore rapid double-tap
    lastTapRef.current = now;
    handler();
  }, [handler, ms]);
}

const onConfirmedPress = useDebouncedPress(onPress, 800);
<Pressable onPress={onConfirmedPress} />

Expo Router (file-based routing)

Expo Router is the file-system-based router for Expo apps. Routes are inferred from file paths under app/.

// File structure:
// app/
//   _layout.tsx              ← root layout (Stack wrapper)
//   index.tsx                ← route: /
//   vehicle/[id].tsx         ← dynamic route: /vehicle/:id
//   (tabs)/
//     _layout.tsx            ← tab layout
//     feed.tsx               ← /feed
//     profile.tsx            ← /profile

// app/_layout.tsx
import { Stack } from 'expo-router';

export default function RootLayout() {
  return (
    <Stack>
      <Stack.Screen name="index" options={{ title: 'Home' }} />
      <Stack.Screen name="vehicle/[id]" options={{ title: 'Vehicle' }} />
    </Stack>
  );
}

// app/vehicle/[id].tsx
import { useLocalSearchParams, Link } from 'expo-router';

export default function VehicleScreen() {
  const { id } = useLocalSearchParams<{ id: string }>();
  return (
    <View>
      <Text>Vehicle {id}</Text>
      <Link href="/">Back home</Link>
    </View>
  );
}

// Programmatic navigation
import { router } from 'expo-router';

router.push(`/vehicle/${id}`);
router.back();
router.replace('/login');

State decision — local vs Zustand vs React Query

State typeWhere it lives
Form input, modal open/closed, currently-selected tabLocal useState — ephemeral, single-component
Theme, current user, app-wide preferences, auth stateZustand (or Context) — shared across screens, not server-derived
Anything that came from a server: vehicles, messages, telemetryReact Query — caching, dedupe, refetch, optimistic all built in
Ephemeral animation values (scroll position, gesture progress)Reanimated shared values — runs on UI thread

Anti-pattern: storing server data in Zustand then manually keeping it in sync with refetches. That's reinventing React Query.

Zustand store

import { create } from 'zustand';

type AuthStore = {
  user: User | null;
  setUser: (user: User | null) => void;
  logout: () => void;
};

export const useAuthStore = create<AuthStore>((set) => ({
  user: null,
  setUser: (user) => set({ user }),
  logout:  () => set({ user: null }),
}));

// Subscribe to a single slice — re-renders only when that slice changes
function Header() {
  const userName = useAuthStore((s) => s.user?.name);
  return <Text>Hi {userName ?? 'Guest'}</Text>;
}
